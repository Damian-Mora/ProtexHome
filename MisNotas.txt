# 1. Construir el proyecto
npm run build

# 2. Crear carpeta temporal y copiar dist
mkdir temp-deploy
cp -r dist/* temp-deploy/

# 3. Cambiar a la rama gh-pages
git checkout -b gh-pages

# 4. Eliminar todo menos la carpeta temp-deploy
rm -rf *
cp -r temp-deploy/* .
rm -rf temp-deploy

# 5. Agregar y commit
git add .
git commit -m "Deploy: Actualización de la PWA"

# 6. Subir a GitHub
git push origin gh-pages

# 7. Volver a la rama main
git checkout main
